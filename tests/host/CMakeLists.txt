cmake_minimum_required(VERSION 3.22)
project(tab5_ui_render C)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "LVGL configuration")
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "")
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "")
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
add_executable(render_ui render_ui.c ../../main/firmware.c ../../main/storage_gate.c ../../managed_components/espressif__cjson/cJSON/cJSON.c)
target_include_directories(render_ui PRIVATE . ../../main ../../managed_components/espressif__cjson/cJSON)
target_link_libraries(render_ui PRIVATE lvgl)
target_link_options(render_ui PRIVATE "-sENVIRONMENT=node" "-sALLOW_MEMORY_GROWTH=1" "-sASSERTIONS=2")
set_target_properties(render_ui PROPERTIES SUFFIX ".cjs")
